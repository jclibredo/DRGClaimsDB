 FUNCTION GET_DASHBOARD RETURN SYS_REFCURSOR IS
    v_result SYS_REFCURSOR;
BEGIN
    OPEN v_result FOR  
        SELECT 
            (SELECT COUNT(*) FROM DRG_SHADOWBILLING.I10) AS I10_TOTAL,
            (SELECT COUNT(*) FROM DRG_SHADOWBILLING.I10VX) AS I10VX_TOTAL,
            (SELECT COUNT(*) FROM DRG_SHADOWBILLING.CCEX) AS CCEX_TOTAL,
            (SELECT COUNT(*) FROM DRG_SHADOWBILLING.DRG) AS DRG_TOTAL,
            (SELECT COUNT(*) FROM DRG_SHADOWBILLING.PDC) AS PDC_TOTAL,
            (SELECT COUNT(*) FROM DRG_SHADOWBILLING.PCOM) AS PCOM_TOTAL,
            (SELECT COUNT(*) FROM DRG_SHADOWBILLING.MDC) AS MDC_TOTAL,
            (SELECT COUNT(*) FROM DRG_SHADOWBILLING.DCV) AS DCV_TOTAL,
            (SELECT COUNT(*) FROM DRG_SHADOWBILLING.AX) AS AX_TOTAL
        FROM DUAL;
     
    RETURN v_result;
END GET_DASHBOARD;
