FUNCTION GET_ICD10_PREMDC(
    p_pdx_code    IN DRG_SHADOWBILLING.I10.CODE%TYPE,
    age_day       IN DRG_SHADOWBILLING.I10.AGEDMIN%TYPE,
    age_min_year  IN DRG_SHADOWBILLING.I10.AGEMIN%TYPE,
    p_patient_sex IN DRG_SHADOWBILLING.I10.SEX%TYPE
) RETURN SYS_REFCURSOR
AS
    pdx_validation SYS_REFCURSOR;
BEGIN
    OPEN pdx_validation FOR  
        SELECT CODE, MDC, PDC, CC, MAINCC, CCROW, HIV_AX, 
               TRAUMA, ACCPDX, SEX, AGEMIN, AGEDMIN, AGEMAX, AGEDUSE 
        FROM   DRG_SHADOWBILLING.I10 
        WHERE  CODE = p_pdx_code 
          AND  AGEMIN <= age_min_year 
          AND  AGEMAX >= age_min_year
          -- Sex logic: Match patient sex OR 'B' for Both
          AND (SEX = p_patient_sex OR SEX = 'B')
          -- Age Use logic: If 'Y', validate age_day; otherwise ignore
          AND (AGEDUSE <> 'Y' OR AGEDMIN <= age_day);

    RETURN pdx_validation; 
END GET_ICD10_PREMDC;

FUNCTION GET_CCL 
(p_pdx IN VARCHAR2,p_sdx IN VARCHAR2,p_dc IN VARCHAR2)
RETURN VARCHAR2 AS 
v_ccl VARCHAR2(1);
v_cc varchar2(10);
v_ccrow number;
v_maincc varchar2(10);
v_dccol number := 0;
v_dcASsign varchar2(10) := 'false';
v_sdx_ex clob;
c_ccex SYS_REFCURSOR;
v_found varchar2(10);
v_cclm varchar2(1000);
TYPE c_sdx_code IS REF CURSOR;
l_sdx_code c_sdx_code;

BEGIN
    SELECT dccol into v_dccol FROM DC WHERE dc = p_dc;

    if LENGTH(p_sdx) != 0 then

        OPEN l_sdx_code FOR SELECT CC, CCROW, maincc FROM DRG_SHADOWBILLING.I10 WHERE CODE = p_sdx;
        LOOP
            FETCH l_sdx_code INTO v_cc, v_ccrow, v_maincc;
            EXIT WHEN l_sdx_code%notfound;
        END LOOP;
        CLOSE l_sdx_code;

        if v_dcASsign = 'false' then
            if v_cc = '1' OR UPPER(v_cc) = 'TRUE' then
                c_ccex := DRGPKGFUNCTION.get_ccex(v_maincc);
                loop
                fetch c_ccex into v_sdx_ex;
                if v_sdx_ex = p_pdx then
                v_found := 'true';
                exit;
                else
                v_found := 'false';
                end if;
                exit when c_ccex%notfound;
                end loop;

                if v_found = 'false' then
                SELECT ccl into v_cclm FROM cclm WHERE ccrow = v_ccrow;

                v_ccl := SUBSTR(v_cclm, v_dccol,1);
                else
                v_ccl := '0';
                end if;                 
            else
            v_ccl := '0';
            end if;

        else
        v_ccl := '0';
        end if;

    else
        v_ccl := '0';

    end if;

    dbms_output.put_line('CC ROW: ' || v_ccrow);
    dbms_output.put_line('DC COL: ' || v_dccol);
    dbms_output.put_line('CCL: ' || v_ccl);
    RETURN v_ccl;
END GET_CCL;

 FUNCTION SORT_CHARS (
        p_string in varchar2
    ) RETURN varchar AS
        v_sorted varchar2(4000);
        v_string varchar2(1);
        type vcArray is
            table of varchar(4000) index by varchar2(1);
        sorted vcArray;
        key varchar2(1);
    BEGIN
        FOR i in 1..length(p_string) LOOP
            v_string := substr(p_string,i,1);
            if ( sorted.exists(v_string) ) THEN sorted(v_string) := sorted(v_string)
                                                                    || v_string;
            ELSE sorted(v_string) := v_string;
            END IF;
        END LOOP;
        v_sorted := '';
        key := sorted.lASt;
        WHILE key IS NOT NULL LOOP
            v_sorted := v_sorted || sorted(key);
            key := sorted.prior(key);
        END LOOP;
        RETURN v_sorted;
    END;
