FUNCTION  GETUSERBYID(puserid IN USERS.USERID%TYPE) RETURN sys_refcursor

PROCEDURE TOKEN_S (
            Message      OUT VARCHAR2,
            Code         OUT NUMBER,
            ptoken      IN SEEKER_HCF_TOKEN.TOKEN%TYPE,
            pdatecreated IN SEEKER_HCF_TOKEN.DATECREATED%TYPE) AS
            pagecount NUMBER;
BEGIN
    -- TODO: Implementation required for PROCEDURE DRGSEEKER.POSTOTP
    SELECT COUNT(TOKEN) into pagecount FROM SEEKER_HCF_TOKEN;
    IF(pagecount=0) THEN
             INSERT INTO DRG_SHADOWBILLING.SEEKER_HCF_TOKEN(TOKEN,DATECREATED) 
             VALUES (ptoken,pdatecreated);
    ELSE
            UPDATE DRG_SHADOWBILLING.SEEKER_HCF_TOKEN SET TOKEN=ptoken,DATECREATED=pdatecreated WHERE DATECREATED IS NOT NULL;
    END IF;
        Message := 'SUCC';
        Code := 00;
    EXCEPTION
    WHEN OTHERS THEN
        Message := 'An error was encounter while updating user token'|| sqlcode|| '-error-'|| sqlerrm;
        Code := 01;
END TOKEN_S;
